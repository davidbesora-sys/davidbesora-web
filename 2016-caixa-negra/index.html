<script>

const photos = [

  "images/IMG_5552_retocada.jpg",

  "images/IMG_5644_retocada.jpg",

  "images/IMG_5649_retocada.jpg",

  "images/IMG_5667_retocada.jpg",

  "images/IMG_5705_retocada.jpg"

];

 

const viewer = document.querySelector(".viewer");

const viewerImage = document.querySelector(".viewer-image");

const counter = document.querySelector(".counter");

const closeButton = document.querySelector(".close");

const previousButton = document.querySelector(".previous");

const nextButton = document.querySelector(".next");

const toggleButton = document.querySelector(".toggle");

 

let current = 0;

let playing = false;

let timer = null;

 

function showPhoto(index) {

  current = (index + photos.length) % photos.length;

  viewerImage.src = photos[current];

  viewerImage.alt = "Caixa Negra — foto " + (current + 1);

  counter.textContent = String(current + 1).padStart(2, "0") + " / 05";

}

 

function openViewer(index) {

  showPhoto(index);

  viewer.classList.add("open");

  document.body.classList.add("viewer-open");

}

 

function closeViewer() {

  viewer.classList.remove("open");

  document.body.classList.remove("viewer-open");

  stopSlideshow();

}

 

function nextPhoto() {

  showPhoto(current + 1);

}

 

function previousPhoto() {

  showPhoto(current - 1);

}

 

function startSlideshow() {

  playing = true;

  toggleButton.textContent = "❚❚";

  timer = setInterval(nextPhoto, 3000);

}

 

function stopSlideshow() {

  playing = false;

  toggleButton.textContent = "▶";

  clearInterval(timer);

  timer = null;

}

 

document.querySelectorAll(".tile").forEach((tile) => {

  tile.addEventListener("click", () => {

  openViewer(Number(tile.dataset.index));

  });

});

 

closeButton.addEventListener("click", closeViewer);

nextButton.addEventListener("click", nextPhoto);

previousButton.addEventListener("click", previousPhoto);

 

toggleButton.addEventListener("click", () => {

  if (playing) {

  stopSlideshow();

  } else {

  startSlideshow();

  }

});

 

document.addEventListener("keydown", (event) => {

  if (!viewer.classList.contains("open")) return;

 

  if (event.key === "Escape") closeViewer();

  if (event.key === "ArrowRight") nextPhoto();

  if (event.key === "ArrowLeft") previousPhoto();

});

 

showPhoto(0);

</script>
